710:    @staticmethod
711:    def _calculate_spread_bps(
712:        snapshot: LiquiditySnapshot,
713:    ) -> Optional[float]:
714:        """
715:        Calculate bid/ask spread in basis points.
716:
717:        spread_bps = ((ask - bid) / midpoint) * 10,000
718:        """
719:
720:        if snapshot.spread_bps is not None:
721:            if snapshot.spread_bps < 0:
722:                return None
723:
724:            return snapshot.spread_bps
725:
726:        if snapshot.bid is None or snapshot.ask is None:
727:            return None
728:
729:        if snapshot.bid <= 0 or snapshot.ask <= 0:
730:            return None
731:
732:        if snapshot.ask < snapshot.bid:
733:            return None
734:
735:        midpoint = (snapshot.bid + snapshot.ask) / 2.0
736:
737:        if midpoint <= 0:
738:            return None
739:
740:        return (
741:            (snapshot.ask - snapshot.bid)
742:            / midpoint
743:            * 10_000.0
744:        )

746:    @staticmethod
747:    def _book_imbalance(
748:        snapshot: LiquiditySnapshot,
749:    ) -> Optional[float]:
750:        """
751:        Order-book imbalance:
752:
753:            (bid_size - ask_size)
754:            ---------------------
755:            (bid_size + ask_size)
756:
757:        Range:
758:            -1 to +1
759:
760:        This is descriptive only. It is NOT interpreted as direction.
761:        """
762:
763:        bid = snapshot.bid_size
764:        ask = snapshot.ask_size
765:
766:        if bid is None or ask is None:
767:            return None
768:
769:        if bid < 0 or ask < 0:
770:            return None
771:
772:        denominator = bid + ask
773:
774:        if denominator <= 0:
775:            return None
776:
777:        return (bid - ask) / denominator

815:        if lqs is not None:
816:

818:                return LiquidityCondition.HEALTHY.value

820:            if lqs >= 60:

826:            return LiquidityCondition.POOR.value

829:        healthy_signals = 0

831:

887:        if isinstance(value, datetime):

890:            text = str(value).strip()
