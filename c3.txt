171:    def find(
172:        self,
173:        query: Optional[ContractQuery] = None,
174:        **filters: Any,
175:    ) -> list[ContractIdentity]:
176:        """
177:        Find contracts using a ContractQuery or explicit filters.
178:
179:        Explicit filters are merged with the query, with explicit
180:        keyword filters taking precedence.
181:        """
182:        if query is not None and not isinstance(
183:            query,
184:            ContractQuery,
185:        ):
186:            raise ContractServiceValidationError(
187:                "query must be a ContractQuery"
188:            )
189:
190:        normalized_query = (
191:            query.normalized()
192:            if query is not None
193:            else ContractQuery()
194:        )
195:
196:        merged = {
197:            "market": normalized_query.market,
198:            "instrument": normalized_query.instrument,
199:            "venue": normalized_query.venue,
200:            "contract_type": normalized_query.contract_type,
201:            "expiry": normalized_query.expiry,
202:            "option_type": normalized_query.option_type,
203:        }
204:
205:        for key, value in filters.items():
206:            if key not in merged:
207:                raise ContractServiceValidationError(
208:                    f"Unknown contract filter: {key}"
209:                )
210:
211:            merged[key] = value
212:
213:        return self.registry.find(**merged)
