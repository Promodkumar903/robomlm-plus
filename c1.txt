151:    def _validate_contract_structure(self) -> None:

168:            if self.strike is None:

342:def _finite_number(
343:    value: Any,
344:    field_name: str,
345:) -> float:
346:    """Validate and normalize a finite numeric value."""
347:    try:
348:        number = float(value)
349:    except (TypeError, ValueError) as exc:
350:        raise ContractIdentityValidationError(
351:            f"{field_name} must be numeric"
352:        ) from exc
353:
354:    if number != number:
355:        raise ContractIdentityValidationError(
356:            f"{field_name} cannot be NaN"
